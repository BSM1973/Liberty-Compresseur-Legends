cmake_minimum_required(VERSION 3.22)
project(CompresseurLegends VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(dsp_tests tests/dsp.cpp)
target_include_directories(dsp_tests PRIVATE src)
add_executable(quality_control tests/quality.cpp)
target_include_directories(quality_control PRIVATE src)
if(MSVC)
 target_compile_options(dsp_tests PRIVATE /utf-8)
 target_compile_options(quality_control PRIVATE /utf-8)
endif()
enable_testing()
add_test(NAME quality_control COMMAND quality_control)
set_tests_properties(quality_control PROPERTIES TIMEOUT 1200)
add_test(NAME dsp_tests COMMAND dsp_tests)
option(BUILD_PLUGIN "Build JUCE plugin" OFF)
if(BUILD_PLUGIN)
 include(FetchContent)
 FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.10 GIT_SHALLOW TRUE)
 FetchContent_MakeAvailable(JUCE)
 set(formats VST3 Standalone)
 if(APPLE)
  list(APPEND formats AU)
 endif()
 juce_add_plugin(CompresseurLegends COMPANY_NAME "Liberty Music Software" BUNDLE_ID com.libertymusicsoftware.compresseurlegends PLUGIN_MANUFACTURER_CODE Lbms PLUGIN_CODE Cleg FORMATS ${formats} PRODUCT_NAME "Compresseur Legends" IS_SYNTH FALSE NEEDS_MIDI_INPUT TRUE NEEDS_MIDI_OUTPUT FALSE COPY_PLUGIN_AFTER_BUILD FALSE)
 target_sources(CompresseurLegends PRIVATE src/Plugin.cpp)
 target_compile_definitions(CompresseurLegends PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
 if(MSVC)
  target_compile_options(CompresseurLegends PRIVATE /utf-8)
 endif()
 target_link_libraries(CompresseurLegends PRIVATE juce::juce_audio_utils juce::juce_dsp juce::juce_recommended_config_flags)
 add_executable(processor_qc tests/processor_qc.cpp)
 target_compile_definitions(processor_qc PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
 target_link_libraries(processor_qc PRIVATE CompresseurLegends juce::juce_audio_utils juce::juce_dsp juce::juce_recommended_config_flags)
 add_test(NAME processor_qc COMMAND processor_qc)
 set_tests_properties(processor_qc PROPERTIES TIMEOUT 180)
endif()
